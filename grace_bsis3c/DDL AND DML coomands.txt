CREATE DATABASE IF NOT EXISTS arcalinbsis3c;

USE arcalinbsis3c;

CREATE TABLE IF NOT EXISTS patient(
		  PatientID INT NOT NULL PRIMARY KEY,
		  LastName VARCHAR(50) NOT NULL,
		  FirstName VARCHAR(50) NOT NULL,
		  DoctorID VARCHAR(50)NOT NULL,
		  UserName VARCHAR(50)NOT NULL,
		  Password VARCHAR(50)NOT NULL
);
CREATE TABLE IF NOT EXISTS Doctor (
		  DocotorID INT NOT NULL PRIMARY KEY,
		  DocoterName VARCHAR(50) NOT NULL,
		  DepartmentID VARCHAR(50) NOT NULL		 
);
CREATE TABLE IF NOT EXISTS department(
		  DepartmentID INT NOT NULL PRIMARY KEY,
		  DepartmantName VARCHAR(50) NOT NULL
);
CREATE TABLE IF NOT EXISTS appoinment(
		  AppoinmentID INT NOT NULL PRIMARY KEY,
		  PatientID VARCHAR(50) NOT NULL,
		  DocotorID VARCHAR(50) NOT NULL,
		  Datw VARCHAR(50)NOT NULL,
		  DoctorName VARCHAR(50)NOT NULL
);
CREATE TABLE IF NOT EXISTS precription(
		  PrecriptionID INT NOT NULL PRIMARY KEY,
		  AppoinmentID VARCHAR(50) NOT NULL,
		  Medicine VARCHAR(50) NOT NULL
);
